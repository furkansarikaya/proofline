# Pinned hook command strings (ADR 0001 section 9, stage 3): every `command` of the three wiring files equals one line.
# <event> is the hook event argument. PreToolUse is pinned apart: a failed start of node blocks (exit 2), as the runtime reads it.
node "${CLAUDE_PLUGIN_ROOT}/adapters/claude/hooks/proofline-hook.mjs" <event> claude
node "${CLAUDE_PLUGIN_ROOT}/adapters/claude/hooks/proofline-hook.mjs" pretool claude || { echo "proofline: the hook did not start (node 22.13 or newer must be on PATH); failing closed" >&2; exit 2; }
node "$CLAUDE_PROJECT_DIR/.agents/hooks/proofline-hook.mjs" <event> claude
if command -v node >/dev/null 2>&1 && test -f "$CLAUDE_PROJECT_DIR/.agents/hooks/proofline-hook.mjs"; then node "$CLAUDE_PROJECT_DIR/.agents/hooks/proofline-hook.mjs" pretool claude; rc=$?; test $rc -eq 0 && exit 0; test $rc -eq 2 || echo "proofline: the hook ended with status $rc; failing closed" >&2; exit 2; fi; test ! -f "$CLAUDE_PROJECT_DIR/.agents/config.yaml" || { echo "proofline: the hook did not start (node 22.13 or newer must be on PATH, and .agents/hooks/proofline-hook.mjs must exist); failing closed" >&2; exit 2; }
node "$(git rev-parse --show-toplevel)/.agents/hooks/proofline-hook.mjs" <event> codex
R=$(git rev-parse --show-toplevel 2>/dev/null); if command -v node >/dev/null 2>&1 && test -f "$R/.agents/hooks/proofline-hook.mjs"; then node "$R/.agents/hooks/proofline-hook.mjs" pretool codex; rc=$?; test $rc -eq 0 && exit 0; test $rc -eq 2 || echo "proofline: the hook ended with status $rc; failing closed" >&2; exit 2; fi; test ! -f "$R/.agents/config.yaml" || { echo "proofline: the hook did not start (node 22.13 or newer must be on PATH, and .agents/hooks/proofline-hook.mjs must exist); failing closed" >&2; exit 2; }
